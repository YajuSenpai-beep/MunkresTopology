% 图 7.1 —— 手工重建（原书 §7 Example 2：Z₊×Z₊ 的“数数”示意）
%%
%% ── 图上有什么 ─────────────────────────────────────────────
%%   左panel：4×4 点阵。a₁..a₁₀ 按 x+y 递增编号（a₁=(0,0) a₂=(1,0) a₃=(0,1) a₄=(2,0) …）。
%%            点阵里画了一批**短斜箭头**，方向一律左上：它们标的是「沿反对角线数下去」的方向，
%%            即从右下的 (c+1,r) 指到左上的 (c,r+1)。箭头的条数按 c 递减：
%%            c=0 有 4 条（r=0..3）、c=1 有 3 条、c=2 有 2 条、c=3 有 1 条 —— 共 10 条。
%%            最靠外的几条（c=0,r=3 等）已经伸出点阵之外，表示计数还要继续。
%%   右panel：A = {(x,y) ∈ Z₊×Z₊ : y ≤ x}。画成**阶梯点阵**：第 c 列有 c+1 个点
%%            （c=0..3，共 1+2+3+4=10 个）。每列在自己的相邻两点之间画一支**向上**的箭头
%%            （列 c 有 c 支），同样表示向上继续。
%%   中间：一条浅弧 + 箭头，标 f —— 表示左panel 到右panel 的那个双射。
%%
%% ── 尺寸从哪来 ─────────────────────────────────────────────
%% 逐点实测 `images\7.1.png`（1150×562 像素，本文件的坐标就是它），并用原书扫描件
%% （`论文素材\3.Munkres原始扫描文本 (Doc2X)\原书PDF\《Topology》 (2nd，James Munkres).pdf`
%%   第 62 页）复核过几何关系。实测值：
%%     · 点阵  列距 105px、行距 99px（原书页坐标 23.0pt × 21.8pt —— 原图本就不是正方格，照抄）
%%     · 圆点  直径 15px；边框线宽 4px；箭头杆 4px；箭头头 17×20px
%%     · 短划占 dot→dot 的 0.22~0.79（两处实测为 0.238/0.797 与 0.19/0.77，取齐成一组）
%%     · f 弧三点拟合（最大偏差 0.31px）：起 (474,259.5)、控制 (520.5,243.6)/(562,247.7)、
%%       终 (592,264)，弧顶 (531,249.5)
%%     · 字号 标签 a = Arial **粗斜体 30pt**；下标数字 = 粗体 22pt；f 与 A = 30pt
%%
\documentclass[12pt]{standalone}
\usepackage{fontspec}
\setsansfont{Arial}
\newfontfamily\mfallback{DejaVu Sans}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\begin{document}
\begin{tikzpicture}[x=1pt, y=-1pt, line width=4pt, line cap=butt, line join=miter]

  %% ════ 两个面板的 L 形边框（竖线 + 底线）════
  \draw (13.8,17) -- (13.8,549.8) -- (514,549.8);
  \draw (635,16) -- (635,549.8) -- (1137,549.8);

  %% ════ 左 panel：4×4 点阵（列距 105、行距 99，起于 (117.5,442)）════
  \foreach \c in {0,...,3} \foreach \r in {0,...,3}
    \fill (117.5+\c*105, 442-\r*99) circle (7.5);

  %% ════ 右 panel：阶梯点阵 A（第 c 列 c+1 个点，列距 103.7，起于 x=744）════
  \foreach \c/\x in {0/744, 1/847.7, 2/951.4, 3/1055.1}
    \foreach \r in {0,...,\c}
      \fill (\x, 442-\r*99) circle (7.5);

  %% ════ 左 panel 的短斜箭头（一律左上）════
  %% 每条都是 (c+1,r) → (c,r+1) 这一段上 0.22~0.79 的一截
  \draw[-{Triangle[width=17pt,length=20pt]}] (199.4,420.2) -- (139.6,363.8);   % c=0 r=0

  \draw[-{Triangle[width=17pt,length=20pt]}] (199.4,321.2) -- (139.6,264.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (199.4,222.2) -- (139.6,165.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (199.4,123.2) -- (139.6,66.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (304.4,420.2) -- (244.6,363.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (304.4,321.2) -- (244.6,264.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (304.4,222.2) -- (244.6,165.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (409.4,420.2) -- (349.6,363.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (409.4,321.2) -- (349.6,264.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (514.4,420.2) -- (454.6,363.8);

  %% ════ 右 panel 的竖直箭头（每列在自己的相邻两点之间）════
  \draw[-{Triangle[width=17pt,length=20pt]}] (847.7,420.2) -- (847.7,363.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (951.4,420.2) -- (951.4,363.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (951.4,321.2) -- (951.4,264.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (1055.1,420.2) -- (1055.1,363.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (1055.1,321.2) -- (1055.1,264.8);
  \draw[-{Triangle[width=17pt,length=20pt]}] (1055.1,222.2) -- (1055.1,165.8);

  %% ════ 中间的 f 弧 ════
  \draw[-{Triangle[width=16pt,length=22pt]}]
    (474,259.5) .. controls (520.5,243.6) and (562,247.7) .. (592,264);
  \node[anchor=center,inner sep=0pt,font=\fontsize{30}{30}\selectfont]
    at (525.5,228) {\textsf{\textbf{\textit{f}}}};

  %% ════ 标签 a₁ … a₁₀ ════
  %% 字号/字重/锚点全部**在原书 600dpi 裁切里实测**（不靠眼估，也不靠自洽指标）：
  %%   量法：原书图区与成品渲染到**同一像素尺度**（原书图区 253×123.5pt ↔ 本画布 1150×562，
  %%         换算 1 原书 pt = 4.5285 本画布单位），在同一个像素框里取连通块比 bbox 与墨迹面积。
  %%   标定：拿圆点当参照 —— 原书 ⌀28px 墨 607，我的 ⌀27px 墨 592，**扫描增益只有 1.03**
  %%         （实心块不膨胀）。所以圆点/线宽这些实心量可以直接比。
  %%   ① 字母 a：原书 15.2×15.8 单位、墨 637；我的常规斜体 30pt 为 14.7×16.3、墨 385，
  %%      粗斜体 30pt 为 14.7×16.3、墨 499。归一化包围盒后 **IoU：常规 0.597 / 粗体 0.768**。
  %%      → 定为 **Arial 粗斜体 30pt**。先前写成 48pt，比原书大 60%（细笔画在印刷扫描里会被
  %%        撑肥，所以原书墨量天然比屏幕渲染重，**不能拿墨量绝对值判字重**，要用 IoU 形状比对）。
  %%   ② 下标数字：原书高 15.8 单位，恰好等于 a 的 x 高；我的 22pt 也是 15.8 —— **高度已对**，
  %%        不必改。数字底边比字母低 18.5 单位（相对圆点中心），是真下标，不是同基线。
  %%   ③ 锚点由**迭代收敛**得到：编译 → 量同框 → 按 (目标−实测) 平移 → 再编译，两轮到位。
  %%        踩过的坑：修正量的**符号**写反过一次，越修越偏；且必须删掉旧 fig.pdf 再编译，
  %%        否则 xelatex 失败时会静默量到上一轮的陈旧 PDF（两版输出一模一样就是这么来的）。
  %% 换算：字母 x 高 px ÷ 0.519 = 字号；数字高 px ÷ 0.716 = 字号。
  \foreach \x/\y/\s/\i in {%
      117.5/442/1/1, 222.5/442/2/2, 327.5/442/4/3, 432.5/442/7/4,

      117.5/244/6/8, 222.5/244/9/9,
      117.5/145/10/10}{

    \node[anchor=base east,inner sep=0pt,font=\fontsize{22}{22}\selectfont]
      at (\x-15.2,\y+20.1) {\textsf{\textbf{\s}}};
  }

  %% ════ 右 panel 的 A 標籤 ════
  \node[anchor=center,inner sep=0pt,font=\fontsize{30}{30}\selectfont]
    at (1102.5,296) {\textsf{\textbf{\textit{A}}}};
