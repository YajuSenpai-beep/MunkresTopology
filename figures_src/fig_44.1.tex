%   （图 81.5 上一度出现 `\draw[line width=7.0pt] (586,277)--(586,270);` 这种 7px 长的疙瘩），

% 做法、约定、已踩过的坑 → `＜工作区＞\checks\redraw\README_重画流程.md`

%%
% ⚠ 本图 tikzpicture 是 y=-1pt（**y 翻转**）⇒ `arc` 的角向"递减"才是屏幕逆时针；
%   同一根线上的多段弧必须同向。改完务必拿原书并排、放大核箭头朝向。

\documentclass[12pt]{standalone}
\usepackage{fontspec}
\setsansfont{Arial}
\newfontfamily\mfallback{DejaVu Sans}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\begin{document}
\begin{tikzpicture}[x=1pt, y=-1pt, line width=4.0pt, line cap=round, line join=round]
  \def\arr{-{Triangle[width=13pt,length=24pt]}}

  % ── 左：带刻度的线段（两端刻度上下对称；中刻度只向上）──
  \draw (25.5,158.5) -- (292.5,158.5);                       % 主线
  \foreach \x/\ya/\yb in {25.5/142.0/174.0, 158.5/143.0/158.5, 292.5/141.0/175.0}
    \draw (\x,\ya) -- (\x,\yb);

  % ── 中：向右的弧形箭头（三次贝塞尔，末端切向 19.5°）──
  \draw[line width=3.8pt,\arr]
    (328.0,131.2) .. controls (358.2,121.6) and (391.5,123.3) .. (423.0,134.5);

  % ── 右：正方形 + 内接三角形 ──
  \draw[line width=4.4pt] (455.5,18.0) rectangle (724.0,292.0);
  \draw[line width=4.0pt] (455.5,292.0) -- (589.0,154.0) -- (724.0,292.0);

  % ── 3 个实心圆点（三角形顶点、方形左下角、右下角）──
  \foreach \p in {(589.0,154.0), (455.5,291.5), (722.0,292.0)} \fill \p circle (8.0);

  % ── 标签（数字不斜体；不用 $...$）──
  \node[anchor=center,inner sep=0pt,font=\fontsize{30.7}{37}\selectfont]
    at (22.0,206.7) {\textsf{\textbf{\textit{a}}}};
  \node[anchor=center,inner sep=0pt,font=\fontsize{30.7}{37}\selectfont]
    at (291.8,203.1) {\textsf{\textbf{\textit{b}}}};
  \node[anchor=center,inner sep=0pt,font=\fontsize{30.7}{37}\selectfont]
    at (371.6,102.4) {\textsf{\textbf{\textit{g}}}};

  % 钉死画布：否则超出的图元/控制点会把页面撑大，1:1 回比就失准
  \pgfresetboundingbox
  \path[use as bounding box] (0,0) rectangle (739,309);
\end{tikzpicture}
\end{document}
